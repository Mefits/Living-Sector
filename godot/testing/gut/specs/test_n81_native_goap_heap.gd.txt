extends GutTest
## N81 pure source planner: the optimized heap MUST preserve the exact
## old deterministic (cost, insertion-order) result. No simulated money.
const N81 = preload("res://scripts/sim/ships/ai/native_goap_planner.gd")

func test_min_heap_exactly_matches_original_sorted_frontier_order() -> void:
	var heap: Array = []
	var original: Array = []
	for index in range(150):
		var entry: Dictionary = {
			"cost": (index * 17 + 9) % 13,
			"order": index,
			"state": {}, "actions": []}
		original.append(entry.duplicate(true))
		N81._frontier_push(heap, entry)
	original.sort_custom(func(a: Dictionary, b: Dictionary) -> bool:
		return int(a["cost"]) < int(b["cost"]) if a["cost"] != b["cost"] else (
			int(a["order"]) < int(b["order"])))
	for expected in original:
		var actual: Dictionary = N81._frontier_pop(heap)
		assert_eq(actual["cost"], expected["cost"])
		assert_eq(actual["order"], expected["order"])
	assert_true(heap.is_empty())

func test_original_trade_goap_yields_paid_physical_route_intents() -> void:
	var start: Dictionary = N81.initial("Origin", 10)
	var registry: Dictionary = {
		"origin": "home-station",
		"source": "source-station",
		"dest": "destination-station",
		"cost_origin_to_source": 1,
		"cost_dest_to_source": 1,
		"commodity": "food",
		"quantity": 1
	}
	var plan: Dictionary = N81.plan(start, "trade", registry, 20, 2)
	assert_false(plan.is_empty(), "Native N81 must find a real trade itinerary")
	if plan.is_empty():
		return
	var kinds: Array = []
	for action in plan["actions"]:
		kinds.append(str(action.get("type", "")))
	assert_eq(kinds, ["Travel", "Dock", "Buy", "Undock", "Travel",
		"Dock", "SellAll"])
	assert_eq(int(plan.get("cost", -1)), 7)

func test_invalid_transition_never_mutates_and_valid_transition_is_detached() -> void:
    var initial_state: Dictionary = N81.initial("Origin", 10)
    var baseline: Dictionary = initial_state.duplicate(true)
    assert_true(N81._apply(initial_state, "Buy", {}).is_empty())
    assert_true(N81._apply(initial_state, "Refuel", {}).is_empty())
    assert_eq(initial_state, baseline)
    var docked: Dictionary = N81._apply(initial_state, "Dock", {})
    assert_false(docked.is_empty())
    if docked.is_empty():
        return
    assert_eq(docked["cost"], 1)
    assert_true(docked["state"]["docked"])
    assert_false(initial_state["docked"])
    var traveling: Dictionary = N81._apply(initial_state,
        "Travel:Source", {
            "origin": "alpha-station", "source": "earth-station",
            "cost_origin_to_source": 3})
    assert_false(traveling.is_empty())
    if traveling.is_empty():
        return
    assert_eq(traveling["cost"], 3)
    assert_eq(traveling["state"]["fuel_bucket"], 7)
    assert_eq(traveling["state"]["at"], "Source")
    assert_eq(initial_state, baseline,
        "Native N81 must never modify original immutable search state")

func test_packed_keys_preserve_all_canonical_n81_distinctions() -> void:
    var keys: Dictionary = {}
    var collisions: int = 0
    var flags: Array[String] = [
        "docked", "cargo_loaded", "delivered",
        "sold", "retrofitted", "rested"]
    for role in ["Origin", "Source", "Dest", "Shipyard"]:
        for fuel in range(11):
            for mask in range(64):
                var state: Dictionary = N81.initial(role, fuel)
                for bit in range(flags.size()):
                    state[flags[bit]] = (mask & (1 << bit)) != 0
                var key: String = N81._key(state)
                if keys.has(key):
                    collisions += 1
                else:
                    keys[key] = true
    assert_eq(collisions, 0, "Distinct native GOAP states must never alias")
    assert_eq(keys.size(), 4 * 11 * 64,
        "N81 min-cost source search must retain 2,816 unique states")
    var exotic: Dictionary = N81.initial("unknown-role", 10)
    assert_eq(N81._key(exotic), JSON.stringify([
        exotic.at, exotic.docked, exotic.fuel_bucket,
        exotic.cargo_loaded, exotic.delivered, exotic.sold,
        exotic.retrofitted, exotic.rested]),
        "Unknown source role still uses its original JSON state identity")
